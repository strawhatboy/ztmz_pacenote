cmake_minimum_required(VERSION 3.24)
project(ZtmzStartupLauncher LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_library(startup_core STATIC startup_core.cpp)
target_link_libraries(startup_core PUBLIC nlohmann_json::nlohmann_json)
target_compile_definitions(startup_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(startup_core PUBLIC /utf-8 /W4)
add_executable(ztmz_launcher WIN32 main.cpp launcher.rc)
set_target_properties(ztmz_launcher PROPERTIES OUTPUT_NAME "ZTMZ.Launcher")
target_link_libraries(ztmz_launcher PRIVATE startup_core comctl32 shell32 ole32 advapi32)
target_link_options(ztmz_launcher PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/launcher.manifest")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/Release/ThirdPartyNotices")
configure_file("${json_SOURCE_DIR}/LICENSE.MIT" "${CMAKE_BINARY_DIR}/Release/ThirdPartyNotices/Launcher-nlohmann-json.txt" COPYONLY)
include(CTest)
if(BUILD_TESTING)
  add_executable(startup_tests tests.cpp)
  target_link_libraries(startup_tests PRIVATE startup_core shell32)
  add_test(NAME startup_policy COMMAND startup_tests)
  # Exercise the same launcher with a harmless child and an isolated mutex;
  # integration tests must never close the user's running ZTMZ instance.
  add_executable(startup_launcher_fixture WIN32 main.cpp launcher.rc)
  target_compile_definitions(startup_launcher_fixture PRIVATE ZTMZ_STARTUP_TEST_MUTEX)
  target_link_libraries(startup_launcher_fixture PRIVATE startup_core comctl32 shell32 ole32 advapi32)
  target_link_options(startup_launcher_fixture PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/launcher.manifest")
  add_executable(startup_child_fixture child_fixture.cpp)
  target_link_libraries(startup_child_fixture PRIVATE startup_core)
endif()
