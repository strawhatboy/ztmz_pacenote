cmake_minimum_required(VERSION 3.24)
project(ZtmzNativeDashboard LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# This list controls implemented native adapters, not merely ImGui backend sources.
if(WIN32)
  set(ZTMZ_DEFAULT_BACKENDS "dx9;dx10;dx11;dx12")
else()
  set(ZTMZ_DEFAULT_BACKENDS "")
endif()
set(ZTMZ_RENDER_BACKENDS "${ZTMZ_DEFAULT_BACKENDS}" CACHE STRING "Native graphics adapters to build (semicolon-separated)")
set(ZTMZ_KNOWN_BACKENDS dx9 dx10 dx11 dx12 vulkan)
foreach(backend IN LISTS ZTMZ_RENDER_BACKENDS)
  if(NOT backend IN_LIST ZTMZ_KNOWN_BACKENDS)
    message(FATAL_ERROR "Unknown graphics backend '${backend}'. Known APIs: ${ZTMZ_KNOWN_BACKENDS}")
  endif()
  if(NOT WIN32)
    message(FATAL_ERROR "Backend '${backend}' requires Windows. Use an empty ZTMZ_RENDER_BACKENDS list for portable tests.")
  endif()
endforeach()
list(REMOVE_DUPLICATES ZTMZ_RENDER_BACKENDS)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)
FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
  URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae)
FetchContent_MakeAvailable(lua)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${lua_SOURCE_DIR}/src/*.c")
list(REMOVE_ITEM LUA_SOURCES "${lua_SOURCE_DIR}/src/lua.c" "${lua_SOURCE_DIR}/src/luac.c")
add_library(ztmz_lua STATIC ${LUA_SOURCES})
target_include_directories(ztmz_lua PUBLIC "${lua_SOURCE_DIR}/src")
set_target_properties(ztmz_lua PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(UNIX)
  target_link_libraries(ztmz_lua PUBLIC m)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lua/host.lua" ZTMZ_LUA_HOST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS lua/host.lua)
configure_file(src/lua_host.hpp.in generated/lua_host.hpp @ONLY)
add_library(ztmz_dashboard_core STATIC src/scene.cpp src/lua_dashboard.cpp src/dashboard_client.cpp)
target_include_directories(ztmz_dashboard_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ztmz_dashboard_core PUBLIC ztmz_lua)
set_target_properties(ztmz_dashboard_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ztmz_dashboard_core PUBLIC src)
target_link_libraries(ztmz_dashboard_core PUBLIC nlohmann_json::nlohmann_json)
find_package(Threads REQUIRED)
include(CTest)
if(BUILD_TESTING)
  add_executable(scene_tests tests/scene_tests.cpp)
  target_link_libraries(scene_tests PRIVATE ztmz_dashboard_core)
  add_test(NAME scene_protocol COMMAND scene_tests)
  add_executable(lua_tests tests/lua_tests.cpp)
  add_test(NAME lua_runtime COMMAND lua_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME builtin_dashboard_lua COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_builtin_dashboards.py"
    $<TARGET_FILE:lua_tests> "${CMAKE_CURRENT_SOURCE_DIR}/../ZTMZ.PacenoteTool.WebDashboard")
  set_tests_properties(scene_protocol lua_runtime builtin_dashboard_lua PROPERTIES TIMEOUT 60)
endif()
FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.91.9b
  URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
FetchContent_MakeAvailable(imgui)
add_library(ztmz_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp src/imgui_context.cpp)
target_include_directories(ztmz_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_include_directories(ztmz_imgui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(ztmz_imgui PUBLIC IMGUI_USER_CONFIG="ztmz_imgui_config.hpp")
target_compile_definitions(ztmz_imgui PUBLIC "ImDrawIdx=unsigned int")
target_compile_definitions(ztmz_imgui PUBLIC IMGUI_USE_WCHAR32)
target_sources(ztmz_dashboard_core PRIVATE src/font_catalog.cpp)
target_link_libraries(ztmz_dashboard_core PUBLIC ztmz_imgui)
if(WIN32)
  target_compile_definitions(ztmz_dashboard_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(ztmz_dashboard_core PRIVATE dwrite)
endif()
set_target_properties(ztmz_imgui PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(ztmz_dashboard_renderer STATIC src/renderer.cpp src/font_atlas.cpp src/font_worker.cpp
  src/graphics/render_api.cpp src/graphics/render_session.cpp)
target_link_libraries(ztmz_dashboard_renderer PUBLIC ztmz_dashboard_core ztmz_imgui Threads::Threads)
set_target_properties(ztmz_dashboard_renderer PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(BUILD_TESTING)
  target_link_libraries(lua_tests PRIVATE ztmz_dashboard_renderer)
endif()
if(BUILD_TESTING)
  add_executable(backend_tests tests/backend_tests.cpp)
  target_link_libraries(backend_tests PRIVATE ztmz_dashboard_renderer)
  add_test(NAME backend_contract COMMAND backend_tests)
  add_test(NAME backend_configuration COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backend_configuration.py"
    "${CMAKE_COMMAND}" "${CMAKE_CURRENT_SOURCE_DIR}")
  set_tests_properties(backend_contract backend_configuration PROPERTIES TIMEOUT 60)
endif()
if(WIN32 AND ZTMZ_RENDER_BACKENDS)
  add_executable(ztmz_compile_imgui_shaders tools/compile_imgui_shaders.cpp)
  target_link_libraries(ztmz_compile_imgui_shaders PRIVATE d3dcompiler)
  include(cmake/PrecompiledImGui.cmake)
  add_library(ztmz_windows_options INTERFACE)
  target_compile_definitions(ztmz_windows_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)

  # CPU networking/image decoding has no dependency on any graphics API.
  add_library(ztmz_transport_winhttp STATIC src/transport_winhttp.cpp)
  target_link_libraries(ztmz_transport_winhttp PUBLIC ztmz_dashboard_core ztmz_windows_options winhttp windowscodecs ole32)

  # Platform input is separate from both ImGui core and its GPU backend.
  add_library(ztmz_imgui_win32 STATIC ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp)
  target_link_libraries(ztmz_imgui_win32 PUBLIC ztmz_imgui ztmz_windows_options dwmapi)

  set(ZTMZ_GPU_TARGETS "")
  set(ZTMZ_GPU_DEFINITIONS "")
  if("dx11" IN_LIST ZTMZ_RENDER_BACKENDS)
    ztmz_precompiled_imgui(dx11 4_0 dx11_imgui)
    add_library(ztmz_backend_dx11 STATIC src/graphics/dx11/backend.cpp
      src/graphics/dx11/dx11_images.cpp src/graphics/dx11/presenter.cpp
      ${dx11_imgui})
    target_link_libraries(ztmz_backend_dx11 PUBLIC ztmz_dashboard_renderer ztmz_windows_options d3d11 dxgi d3dcompiler)
    list(APPEND ZTMZ_GPU_TARGETS ztmz_backend_dx11)
    list(APPEND ZTMZ_GPU_DEFINITIONS ZTMZ_HAS_DX11=1)
  endif()
  foreach(api IN ITEMS dx9 dx10 dx12)
    if("${api}" IN_LIST ZTMZ_RENDER_BACKENDS)
      set(api_imgui "${imgui_SOURCE_DIR}/backends/imgui_impl_${api}.cpp")
      if(api STREQUAL "dx10")
        ztmz_precompiled_imgui(${api} 4_0 api_imgui)
      elseif(api STREQUAL "dx12")
        ztmz_precompiled_imgui(${api} 5_0 api_imgui)
      endif()
      add_library(ztmz_backend_${api} STATIC src/graphics/${api}/backend.cpp ${api_imgui})
      if(api STREQUAL "dx10")
        target_sources(ztmz_backend_${api} PRIVATE src/graphics/dx10/dx10_images.cpp src/graphics/dx10/presenter.cpp)
      elseif(api STREQUAL "dx12")
        target_sources(ztmz_backend_${api} PRIVATE src/graphics/dx12/presenter.cpp src/hooks/dx12_queue.cpp src/hooks/dx12_late_queue.cpp)
        target_link_libraries(ztmz_backend_${api} PRIVATE minhook)
      endif()
      string(REPLACE "dx" "d3d" native_library "${api}")
      target_link_libraries(ztmz_backend_${api} PUBLIC ztmz_dashboard_renderer ztmz_windows_options ${native_library} dxgi d3dcompiler)
      list(APPEND ZTMZ_GPU_TARGETS ztmz_backend_${api})
      string(TOUPPER "${api}" upper_api)
      list(APPEND ZTMZ_GPU_DEFINITIONS "ZTMZ_HAS_${upper_api}=1")
    endif()
  endforeach()
  if("vulkan" IN_LIST ZTMZ_RENDER_BACKENDS)
    # Load Vulkan only at runtime. A DX game must not require vulkan-1.dll to
    # load our overlay, and developers do not need a separately installed SDK.
    FetchContent_Declare(vulkan_headers URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/tar.gz/refs/tags/v1.3.290
      URL_HASH SHA256=f38a653bf93cab7a2a229a53d2d53b1cba9a2819e4c0a7de13c54085bde9bcf5)
    FetchContent_MakeAvailable(vulkan_headers)
    set(VOLK_PULL_IN_VULKAN OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(volk URL https://codeload.github.com/zeux/volk/tar.gz/refs/tags/vulkan-sdk-1.3.290.0
      URL_HASH SHA256=bb6a6d616c0f2bbd5d180da982a6d92a0948581cec937de69f17883980c6ca06)
    FetchContent_MakeAvailable(volk)
    target_link_libraries(volk PUBLIC Vulkan::Headers)
    target_compile_definitions(volk PUBLIC VK_USE_PLATFORM_WIN32_KHR)
    add_library(ztmz_backend_vulkan STATIC src/graphics/vulkan/backend.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp)
    target_link_libraries(ztmz_backend_vulkan PUBLIC ztmz_dashboard_renderer ztmz_windows_options volk)
    target_compile_definitions(ztmz_backend_vulkan PUBLIC IMGUI_IMPL_VULKAN_USE_VOLK)
    list(APPEND ZTMZ_GPU_TARGETS ztmz_backend_vulkan)
    list(APPEND ZTMZ_GPU_DEFINITIONS ZTMZ_HAS_VULKAN=1)
  endif()
  target_compile_definitions(ztmz_dashboard_renderer PRIVATE ${ZTMZ_GPU_DEFINITIONS})

  FetchContent_Declare(minhook URL https://codeload.github.com/TsudaKageyu/minhook/tar.gz/refs/tags/v1.3.4
    URL_HASH SHA256=1aebeae4ca898330c507860acc2fca2eb335fe446a3a2b8444c3bf8b2660a14e)
  FetchContent_MakeAvailable(minhook)
  add_library(ztmz_hooks_win32 STATIC src/hooks/registry_win32.cpp src/hooks/dxgi_hook.cpp)
  if("dx9" IN_LIST ZTMZ_RENDER_BACKENDS)
    target_sources(ztmz_hooks_win32 PRIVATE src/hooks/dx9_hook.cpp)
  endif()
  if("vulkan" IN_LIST ZTMZ_RENDER_BACKENDS)
    target_sources(ztmz_hooks_win32 PRIVATE src/hooks/vulkan_hook.cpp)
  endif()
  target_compile_definitions(ztmz_hooks_win32 PRIVATE ${ZTMZ_GPU_DEFINITIONS})
  target_link_libraries(ztmz_hooks_win32 PUBLIC ${ZTMZ_GPU_TARGETS} ztmz_windows_options PRIVATE minhook d3d11 dxgi)

  add_library(ztmz_dashboard_overlay SHARED src/dll_win32.cpp src/control_pipe_win32.cpp src/overlay_version.rc)
  target_compile_definitions(ztmz_dashboard_overlay PRIVATE ${ZTMZ_GPU_DEFINITIONS})
  if(BUILD_TESTING AND "dx12" IN_LIST ZTMZ_RENDER_BACKENDS)
    add_executable(dx12_association_probe tests/dx12_association_probe.cpp)
    target_link_libraries(dx12_association_probe PRIVATE ztmz_backend_dx12 ztmz_windows_options d3d12 dxgi)
    add_executable(dx12_capture_host tests/dx12_capture_host.cpp)
    target_link_libraries(dx12_capture_host PRIVATE ztmz_windows_options d3d12 dxgi)
    add_dependencies(dx12_capture_host ztmz_dashboard_overlay)
    add_test(NAME dx12_capture_inactive COMMAND dx12_capture_host $<TARGET_FILE:ztmz_dashboard_overlay>)
    set_tests_properties(dx12_capture_inactive PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  endif()
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_executable(ztmz_dashboard_injector src/injector_win32.cpp)
    target_link_libraries(ztmz_dashboard_injector PRIVATE ztmz_windows_options)
    add_dependencies(ztmz_dashboard_overlay ztmz_dashboard_injector)
  endif()
  if(BUILD_TESTING AND "dx9" IN_LIST ZTMZ_RENDER_BACKENDS)
    add_executable(dx9_injection_host tests/dx9_injection_host.cpp)
    target_link_libraries(dx9_injection_host PRIVATE ztmz_windows_options d3d9)
  endif()
  set_target_properties(ztmz_dashboard_overlay PROPERTIES PREFIX "")
  if("vulkan" IN_LIST ZTMZ_RENDER_BACKENDS)
    add_custom_command(TARGET ztmz_dashboard_overlay POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ztmz_dashboard_overlay>/licenses"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${volk_SOURCE_DIR}/LICENSE.md"
        "$<TARGET_FILE_DIR:ztmz_dashboard_overlay>/licenses/volk.txt")
  endif()
  target_link_libraries(ztmz_dashboard_overlay PRIVATE ztmz_transport_winhttp ztmz_hooks_win32 advapi32)
  if("dx11" IN_LIST ZTMZ_RENDER_BACKENDS)
    add_executable(ztmz_dashboard_viewer src/viewer_win32.cpp)
    target_link_libraries(ztmz_dashboard_viewer PRIVATE ztmz_transport_winhttp ztmz_backend_dx11 ztmz_imgui_win32)
    if(BUILD_TESTING)
      add_library(ztmz_dashboard_overlay_test SHARED src/dll_win32.cpp src/control_pipe_win32.cpp tests/hook_transport_stub.cpp)
      set_target_properties(ztmz_dashboard_overlay_test PROPERTIES PREFIX "")
      target_compile_definitions(ztmz_dashboard_overlay_test PRIVATE ZTMZ_NATIVE_HOOK_TEST)
      target_compile_definitions(ztmz_dashboard_overlay_test PRIVATE ${ZTMZ_GPU_DEFINITIONS})
      target_link_libraries(ztmz_dashboard_overlay_test PRIVATE ztmz_hooks_win32 advapi32)
      if("vulkan" IN_LIST ZTMZ_RENDER_BACKENDS)
        add_executable(vulkan_hook_host tests/vulkan_hook_host.cpp)
        target_link_libraries(vulkan_hook_host PRIVATE ztmz_windows_options volk nlohmann_json::nlohmann_json)
        add_dependencies(vulkan_hook_host ztmz_dashboard_overlay_test)
        add_test(NAME vulkan_hook_lifecycle COMMAND vulkan_hook_host $<TARGET_FILE:ztmz_dashboard_overlay_test>)
        set_tests_properties(vulkan_hook_lifecycle PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
      endif()
      foreach(legacy IN ITEMS dx9 dx10)
        if("${legacy}" IN_LIST ZTMZ_RENDER_BACKENDS)
          add_executable(${legacy}_hook_host tests/legacy_hook_host.cpp)
          string(REPLACE "dx" "d3d" legacy_library "${legacy}")
          target_link_libraries(${legacy}_hook_host PRIVATE ztmz_windows_options ${legacy_library} dxgi)
          if(legacy STREQUAL "dx9")
            target_compile_definitions(${legacy}_hook_host PRIVATE TEST_DX9)
          endif()
          add_dependencies(${legacy}_hook_host ztmz_dashboard_overlay_test)
          add_test(NAME ${legacy}_hook_lifecycle COMMAND ${legacy}_hook_host $<TARGET_FILE:ztmz_dashboard_overlay_test>)
          set_tests_properties(${legacy}_hook_lifecycle PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
        endif()
      endforeach()
      if("dx12" IN_LIST ZTMZ_RENDER_BACKENDS)
        add_executable(dx12_hook_host tests/dx12_hook_host.cpp)
        target_link_libraries(dx12_hook_host PRIVATE ztmz_windows_options d3d12 dxgi version)
        add_dependencies(dx12_hook_host ztmz_dashboard_overlay_test)
        add_test(NAME dx12_hook_late COMMAND dx12_hook_host $<TARGET_FILE:ztmz_dashboard_overlay_test>)
        set_tests_properties(dx12_hook_late PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
        set(ZTMZ_TEST_STREAMLINE_LEGACY_DLL "" CACHE FILEPATH "Optional installed Streamline 1.x DLL for DX12 proxy regression")
        if(ZTMZ_TEST_STREAMLINE_LEGACY_DLL)
          add_test(NAME dx12_hook_streamline_legacy COMMAND dx12_hook_host
            $<TARGET_FILE:ztmz_dashboard_overlay_test> hardware "${ZTMZ_TEST_STREAMLINE_LEGACY_DLL}")
          set_tests_properties(dx12_hook_streamline_legacy PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
        endif()
      endif()
      add_executable(dx11_hook_host tests/dx11_hook_host.cpp)
      target_link_libraries(dx11_hook_host PRIVATE ztmz_windows_options d3d11 dxgi nlohmann_json::nlohmann_json)
      target_include_directories(dx11_hook_host PRIVATE src)
      add_dependencies(dx11_hook_host ztmz_dashboard_overlay_test)
      add_test(NAME dx11_hook_late COMMAND dx11_hook_host $<TARGET_FILE:ztmz_dashboard_overlay_test>)
      add_test(NAME dx11_hook_early COMMAND dx11_hook_host $<TARGET_FILE:ztmz_dashboard_overlay_test> early)
      set_tests_properties(dx11_hook_late dx11_hook_early PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    endif()
  endif()
  foreach(target IN ITEMS ztmz_dashboard_overlay ztmz_dashboard_viewer ztmz_dashboard_overlay_test)
    if(TARGET ${target})
      if(CMAKE_SIZEOF_VOID_P EQUAL 4 AND NOT target STREQUAL "ztmz_dashboard_viewer")
        target_link_options(${target} PRIVATE "/EXPORT:ZtmzDashboardStart=_ZtmzDashboardStart@4" "/EXPORT:ZtmzDashboardStop=_ZtmzDashboardStop@0" "/EXPORT:ZtmzDashboardPrepare=_ZtmzDashboardPrepare@4")
        if(target STREQUAL "ztmz_dashboard_overlay_test")
          target_link_options(${target} PRIVATE "/EXPORT:ZtmzDashboardTestActiveApi=_ZtmzDashboardTestActiveApi@0" "/EXPORT:ZtmzDashboardTestScene=_ZtmzDashboardTestScene@0" "/EXPORT:ZtmzDashboardTestTextures=_ZtmzDashboardTestTextures@4")
        endif()
      endif()
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/../ZTMZ.PacenoteTool.WebDashboard/fonts" "$<TARGET_FILE_DIR:${target}>/fonts")
    endif()
  endforeach()
endif()
